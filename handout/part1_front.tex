\heading{Part 1: build a project you can resume}{Summary and Workshop 1 \hfill 28 September 2026}
\topic{What we covered}
\begin{itemize}
  \item Agents can inspect files, edit code, run checks, and use Git history.
  \item Git records changes. Short \code{AGENTS.md} or \code{CLAUDE.md} files give agents project context.
  \item Work in order: \textbf{inspect $\rightarrow$ run $\rightarrow$ audit $\rightarrow$ save a baseline $\rightarrow$ refactor $\rightarrow$ test $\rightarrow$ extend}.
  \item Verify the agent's claims and diffs yourself. Record commands, results, and open questions for the next researcher.
\end{itemize}
\topic{The reactor case}
\begin{itemize}
  \item An inherited cooled-reactor notebook may find several steady states at one coolant temperature.
  \item Check solver convergence and residuals. A repeatable result does not prove model validity, stability, or that every root was found.
\end{itemize}

\topic{Workshop 1: start from the notebook ZIP}
\textbf{1. Create a private repository (0--10 min)}
\begin{itemize}
  \item \href{https://raw.githubusercontent.com/dowlinglab/claude-for-researchers/main/resources/workshops/cstr/workshop1_notebook.zip}{Download the ZIP}. Unzip it; open the \code{cstr-project/} folder.
  \item Create an empty \textbf{private} GitHub repository named \code{cstr-project}. Copy its URL.
\end{itemize}
\code{git init}\quad\code{git branch -M main}\\
\code{git add README.md notebooks data results/.gitkeep}\\
\code{git commit -m "Import inherited reactor notebook"}\\
\code{git remote add origin YOUR-PRIVATE-REPOSITORY-URL}\\
\code{git push -u origin main}\quad\code{git switch -c workshop1}
\begin{itemize}
  \item Check \code{git log -1 --oneline}, \code{git remote -v}, and \code{git status --short}. If push fails, continue locally and record the blocker.
\end{itemize}

\textbf{2. Build the environment and run (10--22 min)}
\begin{itemize}
  \item If \code{conda --version} fails, \href{https://www.anaconda.com/download/success?reg=skipped-miniconda}{install Miniconda} and reopen the terminal.
  \item Draft \code{environment.yml} from notebook imports. Check it for NumPy, SciPy, pandas, matplotlib, JupyterLab, and pytest.
  \item Fallback: \code{resources/workshops/cstr/environment.yml}. Remove \code{-e .} until you have a package.
\end{itemize}
\code{conda env create -f environment.yml}\\
\code{conda activate cstr-workshop}\quad\code{jupyter lab}
\begin{itemize}
  \item Open \code{cstr-project/} as the workspace. Launch Jupyter there; select its kernel and \textbf{Restart Kernel and Run All Cells}.
  \item Check for \code{results/steady_states.csv} and \code{results/steady_state_locus.png}.
\end{itemize}

% Colour
%
% NDBlue and NDGold are defined by the theme, but the theme's gold
% (RGB 174,143,64) is NOT the University's Dome Gold. Override it.
% Verified 2026-09-02 against the branding pages; re-confirm before the talk,
% since onmessage.nd.edu could not be read directly.
%
%   ND Blue     #0C2340   PMS 289   contrast on white 15.79:1
%   Dome Gold   #C99700   PMS 117   contrast on white  2.65:1  <- ACCENT ONLY
%   Irish Green #00843D   PMS 348   contrast on white  4.81:1
%
% Dome Gold must never carry text on white: 2.65:1 fails WCAG (needs 4.5:1).
% Use it for rules, fills, and the active lifecycle stage. On ND Blue it is
% fine (5.95:1).
% ---------------------------------------------------------------------
\definecolor{NDGold}{HTML}{C99700}      % official Dome Gold, overriding theme

\definecolor{IrishGreen}{HTML}{00843D}

% ---------------------------------------------------------------------
% Body text: exactly two sizes, per slides/style_guide.md (a 2026-09-02 audit

% ---------------------------------------------------------------------
% The recurring motif: the talk's five-beat structure, with one lit.
% Usage: \actstrip{3}  (0 = none lit, for the roadmap/overview slide)
%
% Renamed from an earlier six-stage "lifecycle" framing (Explore, Ground,
% Build, Record, Write, Verify) when the talk restructured around a three-act
% story (Tinker / Understand / Build / Challenge / Trust). Kept the macro
% shape so \stagedivider below didn't need to change. Old name \lifecycle is
% aliased to it so the two calls already in sections/00_open.tex still
% compile; retire the alias once that file is reconciled with the new
% structure (tracked in slides/storyboard.md).
% ---------------------------------------------------------------------
\newcommand{\actstrip}[1]{%
\begin{tikzpicture}[x=2.3cm, font=\scriptsize\sffamily]
  \foreach \n [count=\i from 1] in {Tinker, Understand, Build, Challenge, Trust} {
    \pgfmathparse{int(\i == #1)}
    \ifnum\pgfmathresult=1
      \node[fill=NDGold, text=black, rounded corners=2pt,
            inner sep=4pt, minimum width=2.0cm, align=center] at (\i,0) {\textbf{\n}};
    \else
      \node[fill=black!6, text=Muted, rounded corners=2pt,
            inner sep=4pt, minimum width=2.0cm, align=center] at (\i,0) {\n};
    \fi
    \ifnum\i<5
      \draw[-{Stealth[length=4pt]}, Muted, thick]
        ($(\i,0)+(1.02cm,0)$) -- ($(\i,0)+(1.24cm,0)$);
    \fi
  }
\end{tikzpicture}}
\newcommand{\lifecycle}[1]{\actstrip{#1}} % TODO(storyboard): remove once 00_open.tex is rewritten

% A stage divider: the act strip with one stage lit, plus the stage's question.
\newcommand{\stagedivider}[3]{%
\begin{frame}[plain]
  \vfill
  \begin{center}
    \actstrip{#1}\\[1.6em]
    {\LARGE\color{NDBlue}\textbf{#2}}\\[0.7em]
    {\large\color{Muted}#3}
  \end{center}
  \vfill
\end{frame}}

% Small ND logo, bottom-left, on every frame from here on -- reproduces
% Alex's own real-talk slide master (see slides/style_guide.md: bottom-left,
% ~10% of slide width, ~1% margin from each edge), not a new stylistic
% choice. A background template, not \logo{}/frametitle's own logo slot,
% because those place it top-right and are suppressed on [plain] frames;
% a background template survives [plain] (only head/foot/sidebar are
% suppressed there), so the stage dividers below get it too. The title
% slide above is exempt -- it already carries a full-size logo of its own,
% and this line runs only for frames that follow it.
%
% Use \setbeamertemplate{background}{...}, NOT \usebackgroundtemplate: that
% compatibility macro sets the *background canvas* template, which the
% outer theme re-registers with its own (empty) default -- silently
% discarding this override with no error. "background" is the template
% actually meant for this (empty by default, drawn on top of the canvas).
% Also needs `remember picture` and a second compile pass (latexmk does
% this automatically) for `current page` to resolve -- on a single pass
% the node's coordinates collapse to (0,0) and nothing renders. Verified
% both failure modes in isolation before landing this.
% ---------------------------------------------------------------------
\newcommand{\ndcornerlogo}{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=south west, xshift=2mm, yshift=1.5mm] at (current page.south west)
      {\includegraphics[width=0.11\paperwidth]{figures/nd_logo.png}};
  \end{tikzpicture}}
\setbeamertemplate{background}{\ndcornerlogo}

% Colour
%
% NDBlue, NDGold, and IrishGreen are defined by the theme itself, at the
% correct official values.
%
%   ND Blue     #0C2340   PMS 289   contrast on white 15.79:1
%   Dome Gold   #C99700   PMS 117   contrast on white  2.65:1  <- ACCENT ONLY
%   Irish Green #00843D   PMS 348   contrast on white  4.81:1
%
% Dome Gold must never carry text on white: 2.65:1 fails WCAG (needs 4.5:1).
% Use it for rules, fills, and the active lifecycle stage. On ND Blue it is
% fine (5.95:1).
% ---------------------------------------------------------------------

% Semantic colours, taken from the Okabe-Ito palette that
% resources/scripts/figure_style/ recommends -- so the deck uses the
% colour-blind-safe palette the seminar tells people to use.
\definecolor{OkGreen}{HTML}{009E73}     % the practice / verified
\definecolor{OkVermillion}{HTML}{D55E00}% the failure mode  (not red: CB-safe)
\definecolor{OkSky}{HTML}{56B4E9}
% Gray, kept only for non-prose UI elements deliberately meant to look
% inactive/de-emphasized (the act-strip's unlit stage boxes and arrows,
% lstlisting code comments) -- never for a sentence someone has to read
% (too low-contrast). See style_guide.md rule 4.
\definecolor{Muted}{HTML}{7A7A7A}
\definecolor{CodeBg}{HTML}{F4F4F2}

% Frametitle color/background and body text color default to NDBlue rather
% than black, via the theme's own textcolor option (set above) and its
% background-aware defaults. Section-file overrides that set an explicit
% color (OkVermillion, Muted, NDGold, etc. via \good/\bad/\unknown/box
% fills) are unaffected either way.
\setbeamertemplate{navigation symbols}{}

% Numbered-circle (\begin{enumerate}) color: IrishGreen rather than the
% theme's own default NDBlue. IrishGreen is already defined by the
% theme itself (always available, not gated behind an option), so this
% is a plain per-deck override -- no theme change needed.
% FOUND BY TESTING: \setbeamercolor{enumerate item}{fg=...} alone does
% NOT recolor the filled circle -- Beamer's "circles" style numbers use
% the "item projected" color (fg=white, bg=item.fg), a separate entity
% enumerate item only feeds indirectly and that the rendered circle
% never actually reads. bg=... here is what the circle's fill reads.
\setbeamercolor{enumerate item}{fg=IrishGreen}
\setbeamercolor{enumerate subitem}{fg=IrishGreen}
\setbeamercolor{item projected}{bg=IrishGreen}

% ---------------------------------------------------------------------
% The recurring motif: the talk's five-beat structure, with one lit.
% Usage: \actstrip{3}  (0 = none lit, for the roadmap/overview slide)
% Only ever called from \stagedivider below -- font size (\large) is tuned
% to match that macro's own subtitle size.
% ---------------------------------------------------------------------
\newcommand{\actstrip}[1]{%
\begin{tikzpicture}[x=3.1cm, font=\large\sffamily]
  \foreach \n [count=\i from 1] in {Tinker, Understand, Build, Challenge, Trust} {
    \pgfmathparse{int(\i == #1)}
    \ifnum\pgfmathresult=1
      \node[fill=NDGold, text=black, rounded corners=3pt,
            inner sep=6pt, minimum width=2.6cm, align=center] at (\i,0) {\textbf{\n}};
    \else
      \node[fill=black!6, text=Muted, rounded corners=3pt,
            inner sep=6pt, minimum width=2.6cm, align=center] at (\i,0) {\n};
    \fi
    \ifnum\i<5
      \draw[-{Stealth[length=10pt]}, Muted, very thick]
        ($(\i,0)+(1.35cm,0)$) -- ($(\i,0)+(1.75cm,0)$);
    \fi
  }
\end{tikzpicture}}

% A stage divider: the act strip with one stage lit, plus the stage's question.
\newcommand{\stagedivider}[3]{%
\begin{frame}[plain]
  \vfill
  \begin{center}
    \actstrip{#1}\\[1.6em]
    {\LARGE\color{NDBlue}\textbf{#2}}\\[0.7em]
    {\large#3}
  \end{center}
  \vfill
\end{frame}}

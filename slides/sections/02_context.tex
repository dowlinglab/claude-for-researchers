% ================= 2. CONTEXT, NOT CONVERSATION (5 min) =================

\begin{frame}{The shift that matters most}
  \vfill
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{center}\textbf{A conversation}\end{center}
      \begin{itemize}
        \item not versioned
        \item not shared with collaborators
        \item not searchable next year
        \item gone when the session ends
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}\textbf{A repository}\end{center}
      \begin{itemize}
        \item versioned
        \item shared by construction
        \item greppable
        \item outlives everyone involved
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    {\large\color{NDBlue}\textbf{The filesystem is the source of truth.\\
     The conversation is not.}}
  \end{center}
  \vfill
\end{frame}

\begin{frame}[fragile]{This is a real commit message}
  \vfill
  \begin{lstlisting}
Trim analysis and plotting to per-run only

  See conversation for the full KEEP/REMOVE inventory
  and cross-dependency sever plan.
  \end{lstlisting}
  \vfill
  \begin{center}
    {\large That conversation no longer exists.}\\[0.7em]
    {\color{Muted}The change is still in the repository.\\
     The reasoning behind it is not.}
  \end{center}
  \vfill
  \takeaway{This is from one of my own best-run projects. It is the easiest mistake in this talk to make.}
\end{frame}

\begin{frame}{Three kinds of context}
  \vfill
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      {\color{NDBlue}\textbf{Persistent}}\\[0.4em]
      {\small Goals, conventions, how to run things.\\[0.5em]
      True across sessions.}\\[0.7em]
      {\scriptsize\color{Muted}$\rightarrow$ a project file}
    \end{column}
    \begin{column}{0.31\textwidth}
      {\color{NDBlue}\textbf{Authoritative}}\\[0.4em]
      {\small The papers, code, data, and results that decide what is true.}\\[0.7em]
      {\scriptsize\color{Muted}$\rightarrow$ point at it, never summarize it}
    \end{column}
    \begin{column}{0.31\textwidth}
      {\color{NDBlue}\textbf{Task}}\\[0.4em]
      {\small What you are asking for right now.}\\[0.7em]
      {\scriptsize\color{Muted}$\rightarrow$ the prompt (save it if it matters)}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{Most frustration with these tools is really the first two kinds of context going missing.}
\end{frame}

\begin{frame}[fragile]{One file, read at the start of every session}
  \vfill
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}
      \begin{lstlisting}
# Project

Membrane transport parameter estimation.

## Where things are
  src/       the model
  results/   written by scripts; never by hand
  configs/   every run is a config file

## How to run
  conda activate diafiltration
  python -m pytest -m "not slow"

## Conventions
  Results are deterministic. A change in a
  number is a finding, not noise.

## Current state
  Working on: replicate weighting
  Resume at: notes/log.md, 2026-08-14
      \end{lstlisting}
    \end{column}
    \begin{column}{0.34\textwidth}
      \small
      Name it neutrally.\\[0.6em]
      \texttt{CLAUDE.md} and \texttt{AGENTS.md}\\
      become three-line pointers to it.\\[1.2em]
      {\color{Muted}Neither tool reads the other's file --- so the content lives
      in one place and the tool files just point.}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{Write down the thing you would otherwise re-explain in every session.}
\end{frame}

\begin{frame}{The honest limit}
  \vfill
  \begin{center}
    {\large Both of my projects that have one of these files\\
     have a \textbf{stale} one.}\\[1.4em]
    {\color{Muted}One of them warns about stale absolute paths,\\
     directly above four stale absolute paths.}
  \end{center}
  \vfill
  \begin{center}
    {\large\color{NDBlue}\textbf{Written instructions are advice, and advice decays.\\
     A rule you actually need enforced needs something that fails.}}
  \end{center}
  \vfill
  \takeaway{A test that exits non-zero. A script that checks. Not a sentence in a file.}
\end{frame}

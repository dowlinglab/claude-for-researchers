\begin{frame}{Task 5 --- audit a manuscript before it's submitted}
  \vfill
  \slidepoint{
  Let's approach these questions \textbf{systematically}:\\
  \begin{itemize}\setlength\itemsep{0.8em}
    \item Do the methods described match what the code actually did?
    \item Are there arithmetic mistakes in any reported number?
    \item Can every reported number be traced back to a results artifact?
    \item Does rerunning the analysis reproduce it, on a different machine?
    \item Do the manuscript's literature claims match the papers themselves?
  \end{itemize}}
  \vfill
\end{frame}

\begin{frame}{Task 5 --- walk it through, one question at a time}
  \ndlogonote{\scriptsize\href{https://github.com/dowlinglab/claude-for-researchers/blob/main/resources/prompts/manuscript_audit.md}{\texttt{resources/prompts/manuscript\_audit.md}}}
  \vfill
  \begin{center}
    Point Claude at the manuscript source files (LaTeX in Overleaf synced via GitHub ;) and the analysis repo.
  \end{center}
  \vspace{0.6em}
  \begin{center}
  \begin{itemize}\small\setlength\itemsep{0.4em}
    \item Ask \textit{one} question from the list
    \item Traceability: grep (search) the repo for the number, report the file --- or say it can't be found
    \item Method: diff (compare) the manuscript's description against what the code actually does
    \item Arithmetic: recompute from the underlying numbers, don't just re-read them
  \end{itemize}
  \end{center}
  \vspace{0.6em}
  \begin{center}
    \slidepoint{The audit is only as good as the \textbf{files} you hand it.}
  \end{center}
  \vspace{1.8cm plus 1fill}
\end{frame}

\begin{frame}{Task 5 --- we can also automate checking references}
  \ndlogonote{\scriptsize\href{https://github.com/dowlinglab/claude-for-researchers/tree/main/resources/scripts/doi_checker}{\texttt{resources/scripts/doi\_checker/}}}
  \vfill
  \begin{center}\small
    Checks every entry in the library's own versioned \texttt{.bib} file --- the one Task 3's corpus-grounded review builds --- against Crossref: does the DOI resolve, and to \textbf{the paper the entry actually describes}. It reports, never edits, and can't tell you a citation supports the claim it's attached to --- that still needs a human.
  \end{center}
  \vspace{0.5em}
  \begin{center}\small
  \begin{itemize}\setlength\itemsep{0.3em}
    \item A perfect title match, proposed for a completely unrelated document --- rejected by hand.
    \item A citation one reference number off, identical until someone actually opened both papers.
  \end{itemize}
  \end{center}
  \vfill
  \begin{center}
    \slidepoint{A citation is a \textbf{claim} too --- \textbf{verified} or \textbf{wrong}, not close enough.}
  \end{center}
  \vspace{1.8cm plus 1fill}
\end{frame}

  \begin{enumerate}\small\setlength\itemsep{0.3em}

    \item Rerun the relevant gate/test. I review the remaining pedagogical choices.
  \end{enumerate}
  \vfill
  \begin{center}
    \slidepoint{Git preserves the project state. Handoff documents preserve the \textbf{context}.}

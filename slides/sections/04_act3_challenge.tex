\begin{frame}{Task 5 --- audit a manuscript before it's submitted}
  \vfill
  \begin{center}
  \begin{itemize}\small\setlength\itemsep{0.4em}
    \item Do the methods described match what the code actually did?
    \item Are there arithmetic mistakes in any reported number?
    \item Can every reported number be traced back to a results artifact?
    \item Does rerunning the analysis reproduce it, on a different machine?
    \item Do the manuscript's literature claims match the papers themselves?
  \end{itemize}
  \end{center}
  \vspace{0.8em}
  \begin{center}
    \slidepoint{Ask these \textbf{separately} ---\\together, they read as an \textbf{attack}, not a method.}
  \end{center}
  \vfill
\end{frame}

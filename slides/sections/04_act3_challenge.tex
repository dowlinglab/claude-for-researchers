\begin{frame}{Merging three questions reads as an attack}
  \vfill
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      \centering
      \textbf{Arithmetic}\\[0.4em]
      {\small Does this number follow from that one?}
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      \textbf{Method}\\[0.4em]
      {\small Does the method match what's described?}
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      \textbf{Transcription}\\[0.4em]
      {\small Does the manuscript say what the file says?}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    \small Separating them is what makes the criticism land as methodology, not as an attack on the paper.
  \end{center}
  \vfill
\end{frame}

\begin{frame}{"Cannot verify" is a result you want to get}
  \vfill
  \begin{center}
    \unknown{\large Cannot verify}
  \end{center}
  \vspace{0.8em}
  \begin{center}
    \small It is not a failure of the audit --- it's the audit finding a real provenance gap. AI should never fill that gap by guessing; a missing source is a bug you just found for free.
  \end{center}
  \vfill
\end{frame}

\begin{frame}{My status report was wrong about eleven items}
  \vfill
  \begin{center}
    \bad{\large "Everything is up to date."}
  \end{center}
  \vspace{0.4em}
  \begin{center}
    \small That was memory, not evidence. A script that actually checked found eleven items it was wrong about.
  \end{center}
  \vfill
  \begin{center}
    \slidepoint{Determine status from evidence, never recollection.}
  \end{center}
  \vfill
\end{frame}

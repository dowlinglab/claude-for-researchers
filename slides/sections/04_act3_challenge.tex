\begin{frame}{Task 5 --- audit a manuscript before it's submitted}
  \vfill
  \slidepoint{
  Let's approach these questions \textbf{systematically}:\\
  \begin{itemize}\setlength\itemsep{0.8em}
    \item Do the methods described match what the code actually did?
    \item Are there arithmetic mistakes in any reported number?
    \item Can every reported number be traced back to a results artifact?
    \item Does rerunning the analysis reproduce it, on a different machine?
    \item Do the manuscript's literature claims match the papers themselves?
  \end{itemize}}
  \vfill
\end{frame}

\begin{frame}{Task 5 --- walk it through, one question at a time}
  \ndlogonote{\scriptsize\href{https://github.com/dowlinglab/claude-for-researchers/blob/main/resources/prompts/manuscript_audit.md}{\texttt{resources/prompts/manuscript\_audit.md}}}
  \vfill

    Point Claude at the manuscript source files (LaTeX in Overleaf synced via GitHub ;) and the analysis repo.
  \end{center}
  \vspace{0.6em}
  \begin{center}
  \begin{itemize}\small\setlength\itemsep{0.4em}
    \item Ask \textit{one} question from the list --- a merged prompt reads as an attack, same as before
    \item Traceability: grep the repo for the number, report the file --- or say it can't be found
    \item Method: diff the manuscript's description against what the code actually does
    \item Arithmetic: recompute from the underlying numbers, don't just re-read them
  \end{itemize}
  \end{center}
  \vspace{0.6em}
  \begin{center}
    \slidepoint{The audit is only as good as the \textbf{files} you hand it.}
  \end{center}
  \vspace{1.8cm plus 1fill}
\end{frame}

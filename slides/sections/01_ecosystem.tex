\begin{frame}{Desktop app, terminal, or editor: choose your workspace}
  \vfill
  \begin{columns}[t]
    \begin{column}{0.32\textwidth}
      \textbf{Desktop app}\\[0.3em]
      {\small No setup, a chat-like window, easy to supervise several tasks at once. Best if you're new to this or juggling parallel work.}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Terminal (CLI)}\\[0.3em]
      {\small Scriptable --- cron jobs, CI, non-interactive runs. Keyboard-only. Best when the terminal is already your workflow.}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Editor extension}\\[0.3em]
      {\small Agent assistance inline in VS Code (or similar) with cursor-aware diffs. Best for a tight edit-review loop in code you're already reading.}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{center}
    \slidepoint{I use the \textbf{desktop app} more than \textbf{90\%} of the time --- supervising several threads at once is the whole point.}
  \end{center}
  \vfill
\end{frame}

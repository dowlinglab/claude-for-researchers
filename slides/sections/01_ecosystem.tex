      \textbf{Context: what can it consult?}
      \begin{itemize}\small\setlength\itemsep{0.35em}
        \item ChatGPT Projects, Claude Projects, NotebookLM
        \item collect instructions, files, and sources
        \item for understanding papers and developing an argument
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Actions: what can it change?}
      \begin{itemize}\small\setlength\itemsep{0.35em}
        \item Codex, Claude Code, Antigravity
        \item inspect files, edit, run checks, use Git
        \item for reproducing a result or revising a document
      \end{itemize}
    \end{column}
  \end{columns}

\begin{frame}{Desktop app, editor, or terminal: choose your workspace}

      \textbf{Desktop app}\\[0.45em]

      \begin{minipage}{\linewidth}

        \item chat-like window
        \item supervise several tasks
      \end{itemize}
      \end{minipage}
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      \textbf{Editor extension}\\[0.45em]
      \placeholderimage{3.9cm}{2.4cm}{VS Code: a proposed diff in the gutter next to the code}\\[0.6em]
      \begin{minipage}{\linewidth}
      \begin{itemize}\small\setlength\itemsep{0.3em}
        \item inline in VS Code
        \item cursor-aware diffs
        \item tight edit-review loop
      \end{itemize}
      \end{minipage}
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      \textbf{Terminal (CLI)}\\[0.45em]
      \placeholderimage{3.9cm}{2.4cm}{Terminal: six to eight lines of a Claude Code session}\\[0.6em]
      \begin{minipage}{\linewidth}
      \begin{itemize}\small\setlength\itemsep{0.3em}
        \item keyboard-only
        \item scriptable: cron, CI
        \item non-interactive runs
      \end{itemize}
      \end{minipage}
    \end{column}
  \end{columns}

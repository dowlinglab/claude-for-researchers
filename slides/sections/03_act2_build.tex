\begin{frame}[fragile]{Best practices: review the diff, verify every revision}
  \vfill
  {\scriptsize A generic, illustrative diff --- not a real commit.}
  \begin{lstlisting}
- df_all = pd.concat([load(f) for f in glob("data/*.csv")])
+ df_run = load(f"data/{run_id}.csv")
  \end{lstlisting}
  \vspace{0.4em}
  \begin{center}
    \small An AI-written summary says ``refactored the pipeline.'' The diff says what actually changed --- read it, not the summary.
  \end{center}
  \vspace{0.6em}
  \begin{center}
    \small \texttt{latexdiff} against the last commit, word-level, catches what a normal re-read misses --- for a proposal, a manuscript, either one.
  \end{center}
  \vspace{0.4em}
  \begin{center}
    {\scriptsize \texttt{resources/scripts/latexdiff\_check.sh}}
  \end{center}

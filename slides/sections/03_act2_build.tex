\begin{frame}[fragile]{Review the diff, not the summary}
  \vfill
  {\scriptsize The same real commit as Act I, viewed as a diff instead of a message.}
  \begin{lstlisting}
- df_all = pd.concat([load(f) for f in glob("data/*.csv")])
- fig = plot_everything(df_all)
+ df_run = load(f"data/{run_id}.csv")
+ fig = plot_run(df_run)
  \end{lstlisting}
  \vfill
  \begin{center}
    \small An AI-written summary says "refactored the analysis pipeline." The diff says it now reads one run instead of globbing every file in the directory --- a fact worth catching before it merges.
  \end{center}

\begin{frame}{Activity 1: inherit a notebook and make it reusable}
  \vfill
  \begin{center}
    \slidepoint{You receive a reactor notebook in a zip file. Can a colleague run, check, and extend it with the chat closed?}\\[1.1em]
    {\small We will walk through both pages of the Part 1 handout now.}\\[0.8em]
    {\small The handoff has a notebook, a parameter file, and no repository, environment, package, or tests.}
  \end{center}
  \vfill
  \ndlogonote{\scriptsize Hands-on: 45 minutes; regroup: 15 minutes}
\end{frame}

\begin{frame}{First, preserve what the inherited notebook does}
  \vfill
  \begin{enumerate}\setlength\itemsep{1em}
    \item Download the handoff, initialize your private Git repository, and commit the original files.
    \item Create a Conda environment. Run all notebook cells from a fresh kernel.
    \item Audit equations, parameters, units, solver checks, CSV, and figure.
    \item Commit the audited outputs \emph{before} moving code into modules.
  \end{enumerate}
  \vfill
\end{frame}

\begin{frame}{Then extract, test, and extend one question}
  \vfill
  \begin{enumerate}\setlength\itemsep{1em}
    \setcounter{enumi}{2}
    \item Move model, solver, sweep, and plotting functions into Python files. Add NumPy-style docstrings and tests; compare with the saved notebook results.
    \item Choose one extension: finer coolant sweep, heat-transfer sensitivity, feed-temperature sensitivity, or solver-guess robustness.
  \end{enumerate}
  \vspace{0.5em}
  \begin{center}\small
    At minute 40, save the gate you reached and its next command. Continue unfinished gates between sessions.
  \end{center}
  \vfill
\end{frame}

\begin{frame}{The handout is the route through the activity}
  \vfill
  \begin{center}

    {\small Ask for help after five minutes stuck. At regroup, bring one checked claim and one open question.}
  \end{center}
  \vfill
\end{frame}

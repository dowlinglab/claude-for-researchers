\begin{frame}{You are done when you can explain what happened}
  \vfill
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\small\setlength\itemsep{0.55em}
        \item I know which folder is the repository root.
        \item I understand what Git tracks and what it ignores.
        \item I have a first commit that records the starting point.
        \item I can say how my local repository differs from GitHub.
        \item The repository is on GitHub.
        \item I can ask Claude to explain the current state.
      \end{itemize}
    \end{column}
    \begin{column}{0.41\textwidth}
      \textbf{Stuck? Ask Claude:}
      \begin{itemize}\small\setlength\itemsep{0.45em}
        \item ``Explain the last step without assuming I know Git.''
        \item ``How could I undo this?''
        \item ``Show me what Git thinks has changed.''
      \end{itemize}
      \vspace{0.6em}
      {\small\textbf{Finished early?} Stay in the same conversation and ask Claude to run and check the notebook.}
    \end{column}
  \end{columns}
  \vspace{1.6cm plus 1fill}
\end{frame}
